Step 6: List, search, filter and sort employees
GET · http://localhost:8080/api/employees?page=1&limit=2

Authorization: employee or admin token.


Method: POST
Endpoint: http://localhost:8080/graphql
{
  "query": "query { employees(page: 1, limit: 2) { total page limit items { id name email salary departmentId } } }"
}


Get all employees
{
  "query": "query { employees(page: 1, limit: 10) { total page limit items { id name email salary departmentId } } }"
}

Get one employee
{
  "query": "query { employee(id: 1) { id name email salary departmentId } }"
}

Filter employees
{
  "query": "query { employees(page: 1, limit: 10, departmentId: 1) { total items { id name departmentId } } }"
}

Sort by salary
{
  "query": "query { employees(page: 1, limit: 10, sortBy: \"salary\", order: \"desc\") { items { id name salary } } }"
}

Get departments
{
  "query": "query { departments { id name description } }"
}
