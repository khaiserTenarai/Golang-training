1. Create the module
--------------------

mkdir employeeapp
cd employeeapp
go mod init employeeapp

2. employee/employee.go
--------------------
package employee

import "fmt"

func DisplayEmployee() {
	fmt.Println("Employee Name: John")
	fmt.Println("Employee ID: 101")
}

3. main.go
--------------
package main

import "employeeapp/employee"

func main() {
	employee.DisplayEmployee()
}

4. Run the program
----------------------

go run .

or 
go run main.go

Output:

Employee Name: Ganesh
Employee ID: 124772
